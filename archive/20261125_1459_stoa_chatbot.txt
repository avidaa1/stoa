var MARCUS_HISTORY=[];
function _mkey(){
  var inp=document.getElementById('marcus-key');
  var k=inp&&inp.value.trim();
  if(k){try{localStorage.setItem('avidaa_anthropic_key',k);}catch(e){}return k;}
  try{var s=localStorage.getItem('avidaa_anthropic_key');if(s){if(inp)inp.value=s;return s;}}catch(e){}
  return '';
}
function marcusForgetKey(){
  try{localStorage.removeItem('avidaa_anthropic_key');}catch(e){}
  var inp=document.getElementById('marcus-key');if(inp)inp.value='';
  MARCUS_HISTORY=[];
  marcusAdd('sage','Your key has been erased from this browser, and our conversation cleared. Nothing of it remains on this device.');
}
window.marcusForgetKey=marcusForgetKey;
function marcusAdd(role,text){
  var log=document.getElementById('marcus-log');
  var cls=role==='user'?'you':'sage';
  var who=role==='user'?'You':'Marcus';
  var div=document.createElement('div');
  div.className='mrow '+cls;
  div.innerHTML='<div class="who">'+who+'</div><div class="bub">'+text.replace(/</g,'&lt;').replace(/\n/g,'<br>')+'</div>';
  log.appendChild(div);log.scrollTop=log.scrollHeight;
  return div;
}
